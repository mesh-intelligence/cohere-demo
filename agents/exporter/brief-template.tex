% (srd013 R6.3): pandoc reads it when md_to_tex converts, and the confined
% engine only ever sees the brief.tex pandoc wrote into the working
% directory. Everything here is declared configuration -- no variable in this
% file is supplied by the document or the request.

% Section headings render as the form's field bars: a full-width black bar
% with the title in white uppercase, the way a message form labels its
% blocks. Subsections keep a lighter ruled style for anything nested.
\titleformat{\section}
  {\normalfont\bfseries}{}{0pt}
  {\SectionBar}
\titlespacing*{\section}{0pt}{14pt}{6pt}
\newcommand{\SectionBar}[1]{%
  \colorbox{black}{\parbox{\dimexpr\textwidth-2\fboxsep\relax}{%
    \textcolor{white}{\small\bfseries\MakeUppercase{#1}}}}}
\titleformat{\subsection}
  {\normalfont\small\bfseries}{}{0pt}
  {\MakeUppercase}[\vspace{-4pt}\rule{\textwidth}{0.4pt}]
\titlespacing*{\subsection}{0pt}{10pt}{4pt}
